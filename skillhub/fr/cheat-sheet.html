Aide-mémoire CLI & Config

Difficulté : Débutant · Temps estimé : 10 min

Référence rapide pour toutes les commandes et configurations utilisées dans cette formation.

CLI de l'automatisation

# Build automatisé complet
proxmox-install --config config.yaml

# Dry run (validation uniquement)
proxmox-install --config config.yaml --dry-run

# Sans GPU passthrough
proxmox-install --config config.yaml --skip-gpu

# Sans création de VM test
proxmox-install --config config.yaml --skip-vm

# Mode debug
proxmox-install --config config.yaml --debug

# Nettoyage uniquement
proxmox-install --config config.yaml --cleanup-only

# Avec fichier de log
proxmox-install --config config.yaml --log-file build.log

Commandes OpenStack

# Créer une instance
openstack server create --image "Debian 12" --flavor "b3-8" \
  --key-name my-key --network Ext-Net proxmox-node

# Afficher une instance
openstack server show proxmox-node

# Supprimer une instance
openstack server delete proxmox-node

Gestion VM Proxmox (qm)

# Créer une VM
qm create 100 --name gpu-vm --memory 8192 --cores 4 \
  --machine q35 --bios ovmf --ostype l26

# Attacher un GPU
qm set 100 --hostpci0 41:00.0,pcie=1,x-vga=1

# CPU pour passthrough
qm set 100 --cpu host,hidden=1,flags=+pcid

# Démarrer/arrêter/statut
qm start 100
qm stop 100
qm status 100

# Détruire la VM
qm destroy 100 --purge

Commandes GPU & IOMMU

# Détecter les GPUs
lspci -nn | grep -iE '(vga|3d|display)'

# Vérifier la liaison driver
lspci -ks 41:00.0

# Statut IOMMU
dmesg | grep -i iommu

# Lister les groupes IOMMU
find /sys/kernel/iommu_groups/ -type l | sort -V

# Modules VFIO chargés
lsmod | grep vfio

# Mettre à jour GRUB
update-grub

# Mettre à jour initramfs
update-initramfs -u -k all
