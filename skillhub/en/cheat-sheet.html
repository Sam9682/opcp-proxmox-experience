CLI & Config Cheat Sheet

Difficulty: Beginner · Estimated time: 10 min

Quick reference for all commands and configurations used in this training.

Automation CLI

# Full automated build
proxmox-install --config config.yaml

# Dry run (validate only)
proxmox-install --config config.yaml --dry-run

# Skip GPU passthrough
proxmox-install --config config.yaml --skip-gpu

# Skip test VM creation
proxmox-install --config config.yaml --skip-vm

# Debug mode
proxmox-install --config config.yaml --debug

# Cleanup only
proxmox-install --config config.yaml --cleanup-only

# With log file
proxmox-install --config config.yaml --log-file build.log

OpenStack Commands

# Create instance
openstack server create --image "Debian 12" --flavor "b3-8" \
  --key-name my-key --network Ext-Net proxmox-node

# Show instance
openstack server show proxmox-node

# Delete instance
openstack server delete proxmox-node

# List flavors
openstack flavor list

# List images
openstack image list | grep -i debian

Proxmox VM Management (qm)

# Create VM
qm create 100 --name gpu-vm --memory 8192 --cores 4 \
  --machine q35 --bios ovmf --ostype l26

# Attach GPU
qm set 100 --hostpci0 41:00.0,pcie=1,x-vga=1

# Set CPU for passthrough
qm set 100 --cpu host,hidden=1,flags=+pcid

# Add disk
qm set 100 --scsi0 local-lvm:32,iothread=1

# Add EFI disk
qm set 100 --efidisk0 local-lvm:1,efitype=4m

# Start/stop/status
qm start 100
qm stop 100
qm status 100

# View configuration
qm config 100

# Destroy VM
qm destroy 100 --purge

# Get next available VMID
pvesh get /cluster/nextid

GPU & IOMMU Commands

# Detect GPUs
lspci -nn | grep -iE '(vga|3d|display)'

# Check driver binding
lspci -ks 41:00.0

# Check IOMMU status
dmesg | grep -i iommu

# List IOMMU groups
find /sys/kernel/iommu_groups/ -type l | sort -V

# Verify VFIO modules loaded
lsmod | grep vfio

# Update GRUB after changes
update-grub

# Update initramfs
update-initramfs -u -k all

Proxmox Service Management

# Check Proxmox version
pveversion

# Service status
systemctl is-active pve-cluster
systemctl is-active pvedaemon
systemctl is-active pveproxy

# Restart web UI
systemctl restart pveproxy

# View Proxmox logs
journalctl -u pvedaemon -f

Configuration File Template

openstack:
  auth_url: "https://auth.cloud.ovh.net/v3"
  region: "GRA9"
  auth_type: "v3applicationcredential"
  application_credential_id: "${OS_APPLICATION_CREDENTIAL_ID}"
  application_credential_secret: "${OS_APPLICATION_CREDENTIAL_SECRET}"

instance:
  base_image_name: "Debian 12"
  flavor_name: "b3-8"
  ssh_username: "debian"
  ssh_key_name: "my-key"
  ssh_key_path: "~/.ssh/id_rsa"

proxmox:
  version: "8.2"
  enterprise_repo: false
  storage_type: "local-lvm"
  network_bridge: "vmbr0"

gpu_passthrough:
  enabled: true
  auto_detect: true
  iommu_type: "intel"
  blacklist_drivers:
    - nouveau
    - nvidia
    - nvidiafb
    - nvidia_drm

test_vm:
  enabled: true
  name: "gpu-test-vm"
  memory_mb: 8192
  cores: 4
  disk_gb: 32
  machine_type: "q35"
  bios: "ovmf"

timeouts:
  instance_ready: 600
  ssh_connection: 300
  proxmox_install: 1200
  reboot_wait: 300
  gpu_setup: 600
  vm_creation: 300
